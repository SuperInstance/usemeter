1//! Token Vault
2//!
3//! Secure local storage for original values that have been redacted.
4//! Maps `[EMAIL_0001]` back to user@example.com
5//! Never leaves the local machine - this is the core of privacy protection.
6//!
7//! # Security Architecture
8//!
9//! The vault is the **only place** where sensitive data exists in plaintext.
10//! All tokens are stored locally in SQLite and never transmitted to the cloud.
11//!
12//! ## Threat Model
13//!
14//! - **Cloud provider**: Cannot access original values (only tokens)
15//! - **Network attacker**: Intercepted tokens are useless without vault
16//! - **Local attacker**: Requires filesystem access to SQLite database
17//! - **Memory dump**: Vault lives in memory while app runs (encrypted swap recommended)
18//!
19//! ## Data Protection
20//!
21//! - Tokens use UUID-style format: `[CATEGORY_NNNN]`
22//! - Counter is global (not per-session) for uniqueness
23//! - Session IDs enable token isolation and cleanup
24//! - SQLite database provides ACID guarantees
25//!
26//! # Thread Safety
27//!
28//! The vault uses `Arc<Mutex<T>>` for thread-safe access:
29//! - Multiple threads can read/write concurrently
30//! - SQLite connection is protected by mutex
31//! - Counters are protected by separate mutex
32//!
33//! # Performance
34//!
35//! - Store: O(1) - Single database INSERT
36//! - Retrieve: O(1) - Single database SELECT with index
37//! - Clear session: O(n) - DELETE WHERE session_id = ?
38//! - Stats: O(n) - GROUP BY query over session
39

57/// The token vault for session-based token storage
58pub struct TokenVault {
59    conn: Arc<Mutex<Connection>>,
60    /// Track counters per category for token generation (global, not per-session)
61    counters: Arc<Mutex<HashMap<String, u32>>>,
62}

64impl TokenVault {
65    /// Create a new vault with a database file
66    ///
67    /// Opens or creates a SQLite database for persistent token storage.
68    /// The database is created with optimized indexes for common queries.
69    ///
70    /// # Arguments
71    /// * `db_path` - Path to SQLite database file (created if doesn't exist)
72    ///
73    /// # Database Schema
74    /// ```sql
75    /// CREATE TABLE tokens (
76    ///     id INTEGER PRIMARY KEY,
77    ///     token TEXT UNIQUE NOT NULL,
78    ///     category TEXT NOT NULL,
79    ///     original TEXT NOT NULL,
80    ///     created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
81    ///     session_id TEXT NOT NULL
82    /// )
83    ///
84    /// CREATE INDEX idx_session_id ON tokens(session_id)
85    /// CREATE INDEX idx_token ON tokens(token)
86    /// ```
87    ///
88    /// # Performance
89    /// - First open: ~10-50ms (creates tables and indexes)
90    /// - Subsequent opens: ~1-5ms (attaches to existing database)
91    pub fn new<P: AsRef<Path>>(db_path: P) -> PrivacyResult<Self> {
92        let conn = Connection::open(db_path)?;
93
94        // Create the tokens table as per Session 12 spec
95        conn.execute(
96            "CREATE TABLE IF NOT EXISTS tokens (
97                id INTEGER PRIMARY KEY,
98                token TEXT UNIQUE NOT NULL,
99                category TEXT NOT NULL,
100                original TEXT NOT NULL,
101                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
102                session_id TEXT NOT NULL
103            )",
104            [],
105        )?;
106
107        // Create index on session_id for efficient cleanup
108        conn.execute(
109            "CREATE INDEX IF NOT EXISTS idx_session_id ON tokens(session_id)",
110            [],
111        )?;
112
113        // Create index on token for efficient lookups
114        conn.execute("CREATE INDEX IF NOT EXISTS idx_token ON tokens(token)", [])?;
115
116        info!("Token vault initialized at database path");
117
118        Ok(Self {
119            conn: Arc::new(Mutex::new(conn)),
120            counters: Arc::new(Mutex::new(HashMap::new())),
121        })
122    }

124    /// Create an in-memory vault (for testing)
125    ///
126    /// Creates a temporary SQLite database in memory.
127    /// Data is lost when the vault is dropped.
128    ///
129    /// # Use Cases
130    /// - Unit tests (fast, no filesystem cleanup)
131    /// - Short-lived processes (no persistent storage needed)
132    /// - Security-sensitive environments (no disk writes)
133    ///
134    /// # Performance
135    /// - Creation: < 1ms (no disk I/O)
136    /// - Operations: Same as file-based vault
137    pub fn in_memory() -> PrivacyResult<Self> {
138        let conn = Connection::open_in_memory()?;
139
140        conn.execute(
141            "CREATE TABLE tokens (
142                id INTEGER PRIMARY KEY,
143                token TEXT UNIQUE NOT NULL,
144                category TEXT NOT NULL,
145                original TEXT NOT NULL,
146                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
147                session_id TEXT NOT NULL
148            )",
149            [],
150        )?;
151
152        conn.execute("CREATE INDEX idx_session_id ON tokens(session_id)", [])?;
153
154        conn.execute("CREATE INDEX idx_token ON tokens(token)", [])?;
155
156        Ok(Self {
157            conn: Arc::new(Mutex::new(conn)),
158            counters: Arc::new(Mutex::new(HashMap::new())),
159        })
160    }

162    /// Store a value and return its token
163    ///
164    /// Generates a unique token for the sensitive value and stores it in the database.
165    /// Tokens are formatted as `[CATEGORY_NNNN]` where NNNN is a zero-padded counter.
166    ///
167    /// # Token Format
168    /// - Category: Alphanumeric + underscores (validated)
169    /// - Counter: Global per category, 4-digit zero-padded (0001-9999)
170    /// - Example: `[EMAIL_0001]`, `[PHONE_0001]`, `[SSN_0001]`
171    ///
172    /// # Arguments
173    /// * `category` - The category of sensitive data (e.g., "EMAIL", "PHONE")
174    /// * `original` - The original sensitive value (can be any string)
175    /// * `session_id` - The session identifier for grouping tokens
176    ///
177    /// # Returns
178    /// The token that can be used to retrieve the original value later
179    ///
180    /// # Errors
181    /// Returns error if:
182    /// - Category is empty or too long (> 50 chars)
183    /// - Category contains invalid characters (not alphanumeric/underscore)
184    /// - Session ID is empty or too long (> 255 chars)
185    /// - Counter overflows (4 billion tokens per category)
186    /// - Database is locked or corrupted
187    ///
188    /// # Performance
189    /// - Time: O(1) - Single INSERT with indexed lookup
190    /// - Memory: O(1) - Fixed-size allocation
191    /// - Locks: Held briefly during counter increment and INSERT
192    #[instrument(skip(self, original), fields(category, session_id))]
193    pub fn store(&self, category: &str, original: &str, session_id: &str) -> PrivacyResult<String> {
194        // Validate category
195        if category.is_empty() {
196            return Err(PrivacyError::Internal(
197                "Category cannot be empty".to_string(),
198            ));
199        }
200
201        if category.len() > MAX_CATEGORY_LENGTH {
202            return Err(PrivacyError::Internal(format!(
203                "Category too long (max {} chars)",
204                MAX_CATEGORY_LENGTH
205            )));
206        }
207
208        if !category.chars().all(|c| c.is_alphanumeric() || c == '_') {
209            return Err(PrivacyError::Internal(
210                "Category must contain only alphanumeric characters and underscores".to_string(),
211            ));
212        }
213
214        // Validate session_id
215        if session_id.is_empty() {
216            return Err(PrivacyError::Internal(
217                "Session ID cannot be empty".to_string(),
218            ));
219        }
220
221        if session_id.len() > MAX_SESSION_ID_LENGTH {
222            return Err(PrivacyError::Internal(format!(
223                "Session ID too long (max {} chars)",
224                MAX_SESSION_ID_LENGTH
225            )));
226        }
227
228        let conn = self
229            .conn
230            .lock()
231            .map_err(|e| PrivacyError::Internal(format!("Lock poisoned: {}", e)))?;
232
233        // Get and increment global counter for this category
234        let mut counters = self
235            .counters
236            .lock()
237            .map_err(|e| PrivacyError::Internal(format!("Lock poisoned: {}", e)))?;
238        let counter = counters.entry(category.to_string()).or_insert(0);
239
240        // Check for overflow before incrementing
241        if *counter == u32::MAX {
242            return Err(PrivacyError::Internal(
243                format!("Token counter overflow for category '{}'. This indicates an excessive number of redactions.", category)
244            ));
245        }
246
247        *counter += 1;
248        let token_number = *counter;
249        drop(counters);
250
251        // Generate token: [CATEGORY_NNNN]
252        let token = format!("[{}_{:04}]", category, token_number);
253
254        debug!(%token, %category, %session_id, "Storing token in vault");
255
256        // Insert into database
257        conn.execute(
258            "INSERT INTO tokens (token, category, original, session_id)
259             VALUES (?1, ?2, ?3, ?4)",
260            params![token, category, original, session_id],
261        )?;
262
263        Ok(token)
264    }

266    /// Retrieve the original value for a token
267    ///
268    /// # Arguments
269    /// * `token` - The token to look up (e.g., "`[EMAIL_0001]`")
270    ///
271    /// # Returns
272    /// * Some(original_value) if token exists
273    /// * None if token not found
274    #[instrument(skip(self), fields(token))]
275    pub fn retrieve(&self, token: &str) -> Option<String> {
276        let conn = match self.conn.lock() {
277            Ok(guard) => guard,
278            Err(poisoned) => {
279                tracing::error!(
280                    %token,
281                    "Mutex is poisoned, attempting to recover from retrieve"
282                );
283                // Attempt to recover by taking the lock
284                poisoned.into_inner()
285            },
286        };
287
288        debug!(%token, "Retrieving token from vault");
289
290        match conn.query_row(
291            "SELECT original FROM tokens WHERE token = ?1",
292            params![token],
293            |row| row.get::<_, String>(0),
294        ) {
295            Ok(original) => {
296                debug!(%token, "Token found in vault");
297                Some(original)
298            },
299            Err(rusqlite::Error::QueryReturnedNoRows) => {
300                debug!(%token, "Token not found in vault");
301                None
302            },
303            Err(e) => {
304                tracing::error!(%token, error = %e, "Error retrieving token from vault");
305                None
306            },
307        }
308    }

310    /// Clear all tokens for a specific session
311    ///
312    /// # Arguments
313    /// * `session_id` - The session to clear
314    ///
315    /// # Returns
316    /// Ok(()) on success, Error on failure
317    #[instrument(skip(self), fields(session_id))]
318    pub fn clear_session(&self, session_id: &str) -> PrivacyResult<()> {
319        let conn = self
320            .conn
321            .lock()
322            .map_err(|e| PrivacyError::Internal(format!("Lock poisoned: {}", e)))?;
323
324        let deleted = conn.execute(
325            "DELETE FROM tokens WHERE session_id = ?1",
326            params![session_id],
327        )?;
328
329        info!(%session_id, deleted, "Cleared session tokens from vault");
330        Ok(())
331    }

333    /// Get statistics about tokens in a session
334    #[instrument(skip(self), fields(session_id))]
335    pub fn session_stats(&self, session_id: &str) -> PrivacyResult<SessionStats> {
336        let conn = self
337            .conn
338            .lock()
339            .map_err(|e| PrivacyError::Internal(format!("Lock poisoned: {}", e)))?;
340
341        let total: i64 = conn.query_row(
342            "SELECT COUNT(*) FROM tokens WHERE session_id = ?1",
343            params![session_id],
344            |row| row.get(0),
345        )?;
346
347        let mut by_category: HashMap<String, i64> = HashMap::new();
348        let mut stmt = conn.prepare(
349            "SELECT category, COUNT(*) FROM tokens WHERE session_id = ?1 GROUP BY category",
350        )?;
351
352        let rows = stmt.query_map(params![session_id], |row| {
353            Ok((row.get::<_, String>(0)?, row.get::<_, i64>(1)?))
354        })?;
355
356        for row in rows {
357            let (category, count) = row?;
358            by_category.insert(category, count);
359        }
360
361        Ok(SessionStats {
362            total_tokens: total as usize,
363            by_category,
364        })
365    }

379    #[test]
380    fn test_store_and_retrieve() {
381        let vault = TokenVault::in_memory().unwrap();
382
383        let token = vault
384            .store("EMAIL", "test@example.com", "session1")
385            .unwrap();
386        assert!(token.contains("EMAIL"));
387        assert_eq!(token, "[EMAIL_0001]");
388
389        let retrieved = vault.retrieve(&token);
390        assert_eq!(retrieved, Some("test@example.com".to_string()));
391    }
